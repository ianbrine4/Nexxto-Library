local function Create(instanceType, properties)
	local instance = Instance.new(instanceType)
	for property, value in pairs(properties or {}) do
		instance[property] = value
	end
	return instance
end

local function Tween(instance, properties, duration, easingStyle, easingDirection)
	duration = duration or 0.3
	easingStyle = easingStyle or Enum.EasingStyle.Quad
	easingDirection = easingDirection or Enum.EasingDirection.Out
	
	local tweenInfo = TweenInfo.new(duration, easingStyle, easingDirection)
	local tween = TweenService:Create(instance, tweenInfo, properties)
	tween:Play()
	return tween
end

function RiftUI:Init(config)
	config = config or {}
	
	local self = setmetatable({}, {__index = RiftUI})
	
	self.Title = config.title or "Rift UI Library"
	self.Subtitle = config.subtitle or "Clean modern Roblox UI"
	self.IconText = config.text or "Rift"
	self.Size = config.size or Vector2.new(600, 450)
	self.AccentColor = config.color or Theme.Accent
	
	self.Tabs = {}
	self.ActiveTab = nil
	self.IsOpen = true
	self.WasOpen = true
	self.IconPosition = nil
	
	self:CreateMainFrame()
	self:CreateDraggableIcon()
	self:CreateProfileSection()
	
	return self
end

function RiftUI:StartPingUpdater()
	spawn(function()
		while self.ScreenGui and self.ScreenGui.Parent do
			local ping = math.random(20, 80)
			self.PingLabel.Text = ping .. " ms"
			
			local pingDot = self.PingLabel:FindFirstChild("PingDot")
			if pingDot then
				if ping < 50 then
					pingDot.BackgroundColor3 = Theme.Success
					self.PingLabel.TextColor3 = Theme.Success
				elseif ping < 100 then
					pingDot.BackgroundColor3 = Theme.Warning
					self.PingLabel.TextColor3 = Theme.Warning
				else
					pingDot.BackgroundColor3 = Theme.Error
					self.PingLabel.TextColor3 = Theme.Error
				end
			end
			
			wait(1)
		end
	end)
end

function RiftUI:HideUI()
	self.WasOpen = true
	self.IsOpen = false
	
	self.IconPosition = self.MainFrame.Position
	
	Tween(self.MainFrame, {
		Size = UDim2.new(0, 0, 0, 0),
		Position = UDim2.new(
			self.IconPosition.X.Scale,
			self.IconPosition.X.Offset + self.Size.X/2,
			self.IconPosition.Y.Scale,
			self.IconPosition.Y.Offset + self.Size.Y/2
		),
	}, 0.4, Enum.EasingStyle.Back, Enum.EasingDirection.In).Completed:Connect(function()
		self.MainFrame.Visible = false
		self.DraggableIcon.Visible = true
		Tween(self.DraggableIcon, {Size = UDim2.new(0, 50, 0, 50)}, 0.3, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
	end)
end

function RiftUI:ShowUI()
	self.IsOpen = true
	
	Tween(self.DraggableIcon, {Size = UDim2.new(0, 0, 0, 0)}, 0.2, Enum.EasingStyle.Back, Enum.EasingDirection.In).Completed:Connect(function()
		self.DraggableIcon.Visible = false
		self.MainFrame.Visible = true
		
		Tween(self.MainFrame, {
			Size = UDim2.new(0, self.Size.X, 0, self.Size.Y),
			Position = self.IconPosition or UDim2.new(0.5, -self.Size.X/2, 0.5, -self.Size.Y/2),
		}, 0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
	end)
end

function RiftUI:MakeDraggable(frame)
	local dragging = false
	local dragInput
	local dragStart
	local startPos
	
	frame.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position
			
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)
	
	frame.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)
	
	UserInputService.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			local delta = input.Position - dragStart
			frame.Position = UDim2.new(
				startPos.X.Scale,
				startPos.X.Offset + delta.X,
				startPos.Y.Scale,
				startPos.Y.Offset + delta.Y
			)
		end
	end)
end

	function tab:AddSlider(config)
		config = config or {}
		local slider = {}
		slider.Value = config.default or config.min or 0
		slider.Min = config.min or 0
		slider.Max = config.max or 100
		
		local frame = Create("Frame", {

		})
		
		local track = Create("Frame", {
			Name = "Track",
			Parent = frame,
			Size = UDim2.new(1, 0, 0, 4),
			Position = UDim2.new(0, 0, 0, 32),

			BorderSizePixel = 0,
		})
		
		local trackCorner = Create("UICorner", {
			Parent = track,
			CornerRadius = UDim.new(1, 0),
		})
		
		local fill = Create("Frame", {
			Name = "Fill",
			Parent = track,
			Size = UDim2.new((slider.Value - slider.Min) / (slider.Max - slider.Min), 0, 1, 0),
			BackgroundColor3 = Theme.SliderFill,
			BorderSizePixel = 0,
		})
		
		local fillCorner = Create("UICorner", {
			Parent = fill,
			CornerRadius = UDim.new(1, 0),
		})
		
		local handle = Create("Frame", {
			Name = "Handle",
			Parent = track,
			Size = UDim2.new(0, 14, 0, 14),
			Position = UDim2.new((slider.Value - slider.Min) / (slider.Max - slider.Min), -7, 0.5, -7),
			BackgroundColor3 = Theme.Text,
			BorderSizePixel = 0,
		})
		
		local handleCorner = Create("UICorner", {
			Parent = handle,
			CornerRadius = UDim.new(1, 0),
		})
		
		local dragging = false
		
		local function updateSlider(input)
			local pos = math.clamp((input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
			slider.Value = math.floor(slider.Min + (pos * (slider.Max - slider.Min)))
			
			Tween(fill, {Size = UDim2.new(pos, 0, 1, 0)}, 0.05)
			Tween(handle, {Position = UDim2.new(pos, -7, 0.5, -7)}, 0.05)
			valueLabel.Text = tostring(slider.Value)
			
			if config.callback then
				config.callback(slider.Value)
			end
		end
		
		handle.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 then
				dragging = true
				Tween(handle, {Size = UDim2.new(0, 18, 0, 18), Position = UDim2.new(handle.Position.X.Scale, -9, 0.5, -9)}, 0.1)
			end
		end)
		
		track.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 then
				updateSlider(input)
				dragging = true
			end
		end)
		
		UserInputService.InputChanged:Connect(function(input)
			if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
				updateSlider(input)
			end
		end)
		
		UserInputService.InputEnded:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 then
				dragging = false
				Tween(handle, {Size = UDim2.new(0, 14, 0, 14), Position = UDim2.new(handle.Position.X.Scale, -7, 0.5, -7)}, 0.1)
			end
		end)
		
		handle.MouseEnter:Connect(function()
			Tween(handle, {BackgroundColor3 = Theme.AccentColor}, 0.2)
		end)
		
		handle.MouseLeave:Connect(function()
			Tween(handle, {BackgroundColor3 = Theme.Text}, 0.2)
		end)
		
		slider.Instance = frame
		slider.SetValue = function(value)
			slider.Value = math.clamp(value, slider.Min, slider.Max)
			local pos = (slider.Value - slider.Min) / (slider.Max - slider.Min)
			Tween(fill, {Size = UDim2.new(pos, 0, 1, 0)}, 0.2)
			Tween(handle, {Position = UDim2.new(pos, -7, 0.5, -7)}, 0.2)
			valueLabel.Text = tostring(slider.Value)
		end
		
		table.insert(self.Elements, slider)
		return slider
	end

						math.clamp((input.Position.Y - saturationFrame.AbsolutePosition.Y) / saturationFrame.AbsoluteSize.Y, 0, 1)

					sat = pos.X

function RiftUI:SwitchTab(newTab)
	if self.ActiveTab == newTab then return end
	
	if self.ActiveTab then
		Tween(self.ActiveTab.Button, {TextColor3 = Theme.TabInactive}, 0.2)
		Tween(self.ActiveTab.Content, {Position = UDim2.new(0.05, 0, 0, 0), Transparency = 1}, 0.2)
		wait(0.1)
		self.ActiveTab.Content.Visible = false
	end
	
	self.ActiveTab = newTab
	
	Tween(newTab.Button, {TextColor3 = Theme.Text}, 0.2)
	
	local indicatorY = newTab.Button.AbsolutePosition.Y - self.TabContainer.AbsolutePosition.Y + 4
	Tween(self.TabIndicator, {Position = UDim2.new(0, 5, 0, indicatorY)}, 0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
	
	newTab.Content.Position = UDim2.new(0.05, 0, 0, 0)
	newTab.Content.Visible = true
	Tween(newTab.Content, {Position = UDim2.new(0, 0, 0, 0)}, 0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
end
