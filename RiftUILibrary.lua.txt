local function Create(className, properties)
	local instance = Instance.new(className)
	for property, value in pairs(properties or {}) do
		instance[property] = value
	end
	return instance
end

local function Tween(instance, tweenInfo, properties)
	local tween = TweenService:Create(instance, tweenInfo, properties)
	tween:Play()
	return tween
end

local function SetStroke(instance, color, thickness)
	local stroke = Create("UIStroke", {
		Color = color or RiftUI.Theme.Stroke,
		Thickness = thickness or 1,
		Parent = instance
	})
	return stroke
end

local function SetCorner(instance, radius)
	local corner = Create("UICorner", {
		CornerRadius = radius or UDim.new(0, 6),
		Parent = instance
	})
	return corner
end

local function SetPadding(instance, padding)
	local pad = Create("UIPadding", {
		PaddingLeft = UDim.new(0, padding or 10),
		PaddingRight = UDim.new(0, padding or 10),
		PaddingTop = UDim.new(0, padding or 10),
		PaddingBottom = UDim.new(0, padding or 10),
		Parent = instance
	})
	return pad
end

local function SetShadow(instance, transparency)
	local shadow = Create("ImageLabel", {
		Name = "Shadow",
		BackgroundTransparency = 1,
		Image = "rbxassetid://1316045217",
		ImageColor3 = Color3.fromRGB(0, 0, 0),
		ImageTransparency = transparency or 0.7,
		ScaleType = Enum.ScaleType.Slice,
		SliceCenter = Rect.new(10, 10, 118, 118),
		Size = UDim2.new(1, 20, 1, 20),
		Position = UDim2.new(0, -10, 0, -10),
		ZIndex = instance.ZIndex - 1,
		Parent = instance
	})
	return shadow
end

local function MakeDraggable(frame, dragArea)
	local dragging = false
	local dragInput, dragStart, startPos
	
	dragArea = dragArea or frame
	
	dragArea.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position
			
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)
	
	dragArea.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)
	
	UserInputService.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			local delta = input.Position - dragStart
			frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		end
	end)
end

		function tab:AddSlider(config)
			config = config or {}
			local text = config.text or "Slider"
			local min = config.min or 0
			local max = config.max or 100
			local default = config.default or min
			local callback = config.callback or function() end
			
			local sliderFrame = Create("Frame", {
				Name = "Slider",

				Size = UDim2.new(0.7, 0, 0, 20),
				TextXAlignment = Enum.TextXAlignment.Left,
				Parent = sliderFrame
			})
			
			local valueLabel = Create("TextLabel", {
				Name = "Value",
				Text = tostring(default),
				Font = Enum.Font.Gotham,
				TextSize = 12,
				TextColor3 = RiftUI.Theme.TextSecondary,
				BackgroundTransparency = 1,
				Size = UDim2.new(0.3, 0, 0, 20),
				Position = UDim2.new(0.7, 0, 0, 0),
				TextXAlignment = Enum.TextXAlignment.Right,
				Parent = sliderFrame
			})
			
			local track = Create("Frame", {
				Name = "Track",
				Size = UDim2.new(1, 0, 0, 4),
				Position = UDim2.new(0, 0, 0, 35),
				BackgroundColor3 = RiftUI.Theme.BackgroundTertiary,
				BorderSizePixel = 0,
				Parent = sliderFrame
			})
			SetCorner(track, UDim.new(1, 0))
			
			local fill = Create("Frame", {
				Name = "Fill",
				Size = UDim2.new((default - min) / (max - min), 0, 1, 0),
				BackgroundColor3 = accentColor,
				BorderSizePixel = 0,
				Parent = track
			})
			SetCorner(fill, UDim.new(1, 0))
			
			local handle = Create("Frame", {
				Name = "Handle",
				Size = UDim2.new(0, 14, 0, 14),
				Position = UDim2.new((default - min) / (max - min), -7, 0.5, -7),
				BackgroundColor3 = RiftUI.Theme.Text,
				BorderSizePixel = 0,
				Parent = track
			})
			SetCorner(handle, UDim.new(1, 0))
			
			local dragging = false
			
			local function UpdateSlider(input)
				local pos = math.clamp((input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
				local value = math.floor(min + (max - min) * pos)
				valueLabel.Text = tostring(value)
				Tween(fill, RiftUI.Animation.Fast, {Size = UDim2.new(pos, 0, 1, 0)})
				Tween(handle, RiftUI.Animation.Fast, {Position = UDim2.new(pos, -7, 0.5, -7)})
				callback(value)
			end
			
			track.InputBegan:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
					dragging = true
					UpdateSlider(input)
				end
			end)
			
			track.InputChanged:Connect(function(input)
				if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
					UpdateSlider(input)
				end
			end)
			
			UserInputService.InputEnded:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
					dragging = false
				end
			end)
			
			return {
				Set = function(value)
					local clamped = math.clamp(value, min, max)
					local pos = (clamped - min) / (max - min)
					valueLabel.Text = tostring(clamped)
					Tween(fill, RiftUI.Animation.Smooth, {Size = UDim2.new(pos, 0, 1, 0)})
					Tween(handle, RiftUI.Animation.Smooth, {Position = UDim2.new(pos, -7, 0.5, -7)})
					callback(clamped)
				end,
				Get = function()
					return tonumber(valueLabel.Text)
				end
			}
		end

				Position = UDim2.new(0, 0, 0, 42),
				BackgroundColor3 = RiftUI.Theme.BackgroundSecondary,
				BorderSizePixel = 0,
				ClipsDescendants = true,
				Visible = false,
				Parent = colorFrame
			})
			SetCorner(pickerFrame, UDim.new(0, 6))
			SetStroke(pickerFrame, RiftUI.Theme.Stroke, 1)
			
			local saturationFrame = Create("Frame", {
				Name = "Saturation",
				Size = UDim2.new(1, -20, 0, 100),
				Position = UDim2.new(0, 10, 0, 10),
				BackgroundColor3 = Color3.fromRGB(255, 0, 0),
				BorderSizePixel = 0,
				Parent = pickerFrame
			})
			SetCorner(saturationFrame, UDim.new(0, 4))
			
			local saturationGradient = Create("UIGradient", {
				Color = ColorSequence.new({
					ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 255, 255)),
					ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 0, 0))
				}),
				Parent = saturationFrame
			})
			
			local darknessFrame = Create("Frame", {
				Name = "Darkness",
				Size = UDim2.new(1, 0, 1, 0),
				BackgroundTransparency = 0,
				BorderSizePixel = 0,
				Parent = saturationFrame
			})
			SetCorner(darknessFrame, UDim.new(0, 4))
			
			local darknessGradient = Create("UIGradient", {
				Rotation = 90,
				Color = ColorSequence.new({
					ColorSequenceKeypoint.new(0, Color3.fromRGB(0, 0, 0)),
					ColorSequenceKeypoint.new(1, Color3.fromRGB(0, 0, 0))
				}),
				Transparency = NumberSequence.new({
					NumberSequenceKeypoint.new(0, 0),
					NumberSequenceKeypoint.new(1, 1)
				}),
				Parent = darknessFrame
			})
			
			local selector = Create("Frame", {
				Name = "Selector",
				Size = UDim2.new(0, 12, 0, 12),
				BackgroundColor3 = RiftUI.Theme.Text,
				BorderSizePixel = 0,
				Position = UDim2.new(0.5, -6, 0.5, -6),
				Parent = saturationFrame
			})
			SetCorner(selector, UDim.new(1, 0))
			SetStroke(selector, Color3.fromRGB(0, 0, 0), 2)
			
			local hueSlider = Create("Frame", {
				Name = "HueSlider",
				Size = UDim2.new(1, -20, 0, 16),
				Position = UDim2.new(0, 10, 0, 118),
				BackgroundColor3 = Color3.fromRGB(255, 255, 255),
				BorderSizePixel = 0,
				Parent = pickerFrame
			})
			SetCorner(hueSlider, UDim.new(0, 4))
			
			local hueGradient = Create("UIGradient", {
				Color = ColorSequence.new({
					ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 0, 0)),
					ColorSequenceKeypoint.new(0.167, Color3.fromRGB(255, 255, 0)),
					ColorSequenceKeypoint.new(0.333, Color3.fromRGB(0, 255, 0)),
					ColorSequenceKeypoint.new(0.5, Color3.fromRGB(0, 255, 255)),
					ColorSequenceKeypoint.new(0.667, Color3.fromRGB(0, 0, 255)),
					ColorSequenceKeypoint.new(0.833, Color3.fromRGB(255, 0, 255)),
					ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 0, 0))
				}),
				Parent = hueSlider
			})
			
			local hueHandle = Create("Frame", {
				Name = "HueHandle",
				Size = UDim2.new(0, 8, 1, 4),
				Position = UDim2.new(0, -4, 0, -2),
				BackgroundColor3 = RiftUI.Theme.Text,
				BorderSizePixel = 0,
				Parent = hueSlider
			})
			SetCorner(hueHandle, UDim.new(0, 2))
			
			local expanded = false
			local hue = 0
			local saturation = 1
			local value = 1
			
			local function UpdateColor()
				local color = Color3.fromHSV(hue, saturation, value)
				preview.BackgroundColor3 = color
				saturationFrame.BackgroundColor3 = Color3.fromHSV(hue, 1, 1)
				callback(color)
			end
			
			local function HandleSaturationInput(input)
				local pos = Vector2.new(
					math.clamp((input.Position.X - saturationFrame.AbsolutePosition.X) / saturationFrame.AbsoluteSize.X, 0, 1),
					math.clamp((input.Position.Y - saturationFrame.AbsolutePosition.Y) / saturationFrame.AbsoluteSize.Y, 0, 1)
				)
				saturation = pos.X
				value = 1 - pos.Y
				Tween(selector, RiftUI.Animation.Fast, {Position = UDim2.new(pos.X, -6, pos.Y, -6)})
				UpdateColor()
			end
			
			local function HandleHueInput(input)
				local pos = math.clamp((input.Position.X - hueSlider.AbsolutePosition.X) / hueSlider.AbsoluteSize.X, 0, 1)
				hue = pos
				Tween(hueHandle, RiftUI.Animation.Fast, {Position = UDim2.new(pos, -4, 0, -2)})
				UpdateColor()
			end
			
			local draggingSat = false
			local draggingHue = false
			
			saturationFrame.InputBegan:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
					draggingSat = true
					HandleSaturationInput(input)
				end
			end)
			
			saturationFrame.InputChanged:Connect(function(input)
				if draggingSat and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
					HandleSaturationInput(input)
				end
			end)
			
			hueSlider.InputBegan:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
					draggingHue = true
					HandleHueInput(input)
				end
			end)
			
			hueSlider.InputChanged:Connect(function(input)
				if draggingHue and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
					HandleHueInput(input)
				end
			end)
			
			UserInputService.InputEnded:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
					draggingSat = false
					draggingHue = false
				end
			end)
			
			preview.MouseButton1Click:Connect(function()
				expanded = not expanded
				if expanded then
					pickerFrame.Visible = true
					colorFrame.Size = UDim2.new(1, 0, 0, 190)
					Tween(pickerFrame, RiftUI.Animation.Smooth, {Size = UDim2.new(1, 0, 0, 150)})
				else
					Tween(pickerFrame, RiftUI.Animation.Smooth, {Size = UDim2.new(1, 0, 0, 0)})
					task.delay(0.3, function()
						pickerFrame.Visible = false
						colorFrame.Size = UDim2.new(1, 0, 0, 40)
					end)
				end
			end)
			
			return {
				Set = function(color)
					local h, s, v = color:ToHSV()
					hue, saturation, value = h, s, v
					preview.BackgroundColor3 = color
					saturationFrame.BackgroundColor3 = Color3.fromHSV(h, 1, 1)
					Tween(selector, RiftUI.Animation.Fast, {Position = UDim2.new(s, -6, 1 - v, -6)})
					Tween(hueHandle, RiftUI.Animation.Fast, {Position = UDim2.new(h, -4, 0, -2)})
					callback(color)
				end,
				Get = function()
					return preview.BackgroundColor3
				end
			}
		end
		
		function tab:AddImage(config)
			config = config or {}
			local imageId = config.image or ""
			local size = config.size or UDim2.new(1, 0, 0, 150)
			local callback = config.callback or function() end
			
			local imageFrame = Create("Frame", {
				Name = "Image",
				Size = size,
				BackgroundColor3 = RiftUI.Theme.BackgroundSecondary,
				BorderSizePixel = 0,
				Parent = self.Page
			})
			SetCorner(imageFrame, UDim.new(0, 8))
			
			local imageLabel = Create("ImageLabel", {
				Name = "ImageLabel",
				Size = UDim2.new(1, -8, 1, -8),
				Position = UDim2.new(0, 4, 0, 4),
				BackgroundTransparency = 1,
				Image = imageId,
				ScaleType = Enum.ScaleType.Fit,
				Parent = imageFrame
			})
			SetCorner(imageLabel, UDim.new(0, 6))
			
			imageFrame.InputBegan:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
					Tween(imageFrame, RiftUI.Animation.Fast, {Size = UDim2.new(size.X.Scale, size.X.Offset - 4, size.Y.Scale, size.Y.Offset - 4), Position = UDim2.new(0, 2, 0, 2)})
				end
			end)
			
			imageFrame.InputEnded:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
					Tween(imageFrame, RiftUI.Animation.Elastic, {Size = size, Position = UDim2.new(0, 0, 0, 0)})
					callback()
				end
			end)
			
			return {
				Set = function(newImageId)
					imageLabel.Image = newImageId
				end,
				Get = function()
					return imageLabel.Image
				end
			}
		end
		
		table.insert(self.Tabs, tab)
		self.TabScroll.CanvasSize = UDim2.new(0, 0, 0, #self.Tabs * 34 + 20)
		
		return tab
	end
	
	function window:SelectTab(tab)
		if self.CurrentTab == tab then return end
		
		if self.CurrentTab then
			Tween(self.CurrentTab.Button, RiftUI.Animation.Fast, {TextColor3 = RiftUI.Theme.TextSecondary, BackgroundColor3 = RiftUI.Theme.Background})
			self.CurrentTab.Page.Visible = false
		end
		
		self.CurrentTab = tab
		Tween(tab.Button, RiftUI.Animation.Fast, {TextColor3 = RiftUI.Theme.Text, BackgroundColor3 = RiftUI.Theme.BackgroundSecondary})
		
		self.Indicator.Visible = true
		Tween(self.Indicator, RiftUI.Animation.Smooth, {Position = UDim2.new(0, 0, 0, tab.Button.AbsolutePosition.Y - self.TabScroll.AbsolutePosition.Y + self.TabScroll.CanvasPosition.Y)})
		
		tab.Page.Visible = true
		tab.Page.Position = UDim2.new(0, 20, 0, 0)
		tab.Page.BackgroundTransparency = 1
		Tween(tab.Page, RiftUI.Animation.Smooth, {Position = UDim2.new(0, 0, 0, 0)})
	end
	
	function window:Notify(config)
		config = config or {}
		local title = config.title or "Notification"
		local text = config.text or ""
		local icon = config.icon or ""
		local duration = config.duration or 3
		local position = config.position or "top-right"
		
		local notificationGui = Create("ScreenGui", {
			Name = "RiftNotification",
			Parent = PlayerGui,
			ResetOnSpawn = false,
			ZIndexBehavior = Enum.ZIndexBehavior.Sibling
		})
		
		local notificationFrame = Create("Frame", {
			Name = "Notification",
			Size = UDim2.new(0, 280, 0, 80),
			BackgroundColor3 = RiftUI.Theme.Background,
			BorderSizePixel = 0,
			Position = position == "top-right" and UDim2.new(1, 300, 0, 20) or
			           position == "top-left" and UDim2.new(0, -300, 0, 20) or
			           position == "bottom-right" and UDim2.new(1, 300, 1, -100) or
			           UDim2.new(0, -300, 1, -100),
			Parent = notificationGui
		})
		SetCorner(notificationFrame, UDim.new(0, 8))
		SetStroke(notificationFrame, RiftUI.Theme.Stroke, 1)
		SetShadow(notificationFrame, 0.6)
		
		local iconLabel = Create("ImageLabel", {
			Name = "Icon",
			Size = UDim2.new(0, 40, 0, 40),
			Position = UDim2.new(0, 15, 0, 20),
			BackgroundTransparency = 1,
			Image = icon,
			Parent = notificationFrame
		})
		
		local titleLabel = Create("TextLabel", {
			Name = "Title",
			Text = title,
			Font = Enum.Font.GothamBold,
			TextSize = 14,
			TextColor3 = RiftUI.Theme.Text,
			BackgroundTransparency = 1,
			Size = UDim2.new(1, -75, 0, 20),
			Position = UDim2.new(0, 65, 0, 15),
			TextXAlignment = Enum.TextXAlignment.Left,
			Parent = notificationFrame
		})
		
		local textLabel = Create("TextLabel", {
			Name = "Text",
			Text = text,
			Font = Enum.Font.Gotham,
			TextSize = 12,
			TextColor3 = RiftUI.Theme.TextSecondary,
			BackgroundTransparency = 1,
			Size = UDim2.new(1, -75, 0, 40),
			Position = UDim2.new(0, 65, 0, 35),
			TextWrapped = true,
			TextXAlignment = Enum.TextXAlignment.Left,
			TextYAlignment = Enum.TextYAlignment.Top,
			Parent = notificationFrame
		})
		
		local closeBtn = Create("TextButton", {
			Name = "Close",
			Text = "X",
			Font = Enum.Font.GothamBold,
			TextSize = 12,
			TextColor3 = RiftUI.Theme.TextSecondary,
			BackgroundTransparency = 1,
			Size = UDim2.new(0, 24, 0, 24),
			Position = UDim2.new(1, -28, 0, 8),
			Parent = notificationFrame
		})
		
		closeBtn.MouseEnter:Connect(function()
			Tween(closeBtn, RiftUI.Animation.Fast, {TextColor3 = RiftUI.Theme.Error})
		end)
		
		closeBtn.MouseLeave:Connect(function()
			Tween(closeBtn, RiftUI.Animation.Fast, {TextColor3 = RiftUI.Theme.TextSecondary})
		end)
		
		local function CloseNotification()
			Tween(notificationFrame, RiftUI.Animation.Smooth, {Position = position:find("right") and UDim2.new(1, 300, notificationFrame.Position.Y.Scale, notificationFrame.Position.Y.Offset) or UDim2.new(0, -300, notificationFrame.Position.Y.Scale, notificationFrame.Position.Y.Offset)})
			task.delay(0.35, function()
				notificationGui:Destroy()
			end)
		end
		
		closeBtn.MouseButton1Click:Connect(CloseNotification)
		
		Tween(notificationFrame, RiftUI.Animation.Elastic, {Position = position:find("right") and UDim2.new(1, -300, notificationFrame.Position.Y.Scale, notificationFrame.Position.Y.Offset) or UDim2.new(0, 20, notificationFrame.Position.Y.Scale, notificationFrame.Position.Y.Offset)})
		
		if duration > 0 then
			task.delay(duration, CloseNotification)
		end
	end
	
	Tween(mainFrame, RiftUI.Animation.Elastic, {Size = UDim2.new(0, size.X, 0, size.Y), Position = UDim2.new(0.5, -size.X/2, 0.5, -size.Y/2)})
	
	return window
end
