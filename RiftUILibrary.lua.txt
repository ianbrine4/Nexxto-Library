local function Tween(instance, info, props)
    local t = TweenService:Create(instance, info, props)
    t:Play()
    return t
end

local function Create(class, props, children)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then
            inst[k] = v
        end
    end
    for _, child in pairs(children or {}) do
        child.Parent = inst
    end
    if props and props.Parent then
        inst.Parent = props.Parent
    end
    return inst
end

local function AddStroke(parent, color, thickness)
    return Create("UIStroke", {
        Parent    = parent,
        Color     = color or COLORS.Border,
        Thickness = thickness or 1,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    })
end

local function AddCorner(parent, radius)
    return Create("UICorner", {
        Parent       = parent,
        CornerRadius = UDim.new(0, radius or 6),
    })
end

local function AddPadding(parent, top, right, bottom, left)
    return Create("UIPadding", {
        Parent          = parent,
        PaddingTop      = UDim.new(0, top or 8),
        PaddingRight    = UDim.new(0, right or 8),
        PaddingBottom   = UDim.new(0, bottom or 8),
        PaddingLeft     = UDim.new(0, left or 8),
    })
end

local function MakeSeparator(parent)
    return Create("Frame", {
        Parent          = parent,
        Size            = UDim2.new(1, 0, 0, 1),
        BackgroundColor3 = COLORS.Border,
        BorderSizePixel = 0,
    })
end

local function HoverEffect(frame, normalColor, hoverColor)
    frame.MouseEnter:Connect(function()
        Tween(frame, EASING.Fast, { BackgroundColor3 = hoverColor })
    end)
    frame.MouseLeave:Connect(function()
        Tween(frame, EASING.Fast, { BackgroundColor3 = normalColor })
    end)
end

local NotificationQueue = {}
local NotificationParent = nil

local function GetNotificationParent()
    if NotificationParent and NotificationParent.Parent then
        return NotificationParent
    end
    local screenGui = Instance.new("ScreenGui")
    screenGui.Name = "RiftNotifications"
    screenGui.ResetOnSpawn = false
    screenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    screenGui.DisplayOrder = 999
    screenGui.Parent = LocalPlayer:WaitForChild("PlayerGui")
    NotificationParent = screenGui
    return screenGui
end

function RiftUI:Notify(options)
    options = options or {}
    local title    = options.title    or "Notification"
    local message  = options.message  or ""
    local duration = options.duration or 4
    local position = options.position or "top-right"
    local icon     = options.icon     or nil

    local parent = GetNotificationParent()
    local cardH = 72
    local cardW = 300
    local pad   = 12

    local anchor, startPos, endPos
    if position == "top-right" then
        anchor   = Vector2.new(1, 0)
        startPos = UDim2.new(1, cardW + 20, 0, pad)
        endPos   = UDim2.new(1, -(cardW + pad), 0, pad)
    elseif position == "top-left" then
        anchor   = Vector2.new(0, 0)
        startPos = UDim2.new(0, -(cardW + 20), 0, pad)
        endPos   = UDim2.new(0, pad, 0, pad)
    elseif position == "bottom-right" then
        anchor   = Vector2.new(1, 1)
        startPos = UDim2.new(1, cardW + 20, 1, -pad)
        endPos   = UDim2.new(1, -(cardW + pad), 1, -(cardH + pad))
    elseif position == "bottom-left" then
        anchor   = Vector2.new(0, 1)
        startPos = UDim2.new(0, -(cardW + 20), 1, -pad)
        endPos   = UDim2.new(0, pad, 1, -(cardH + pad))
    end

    local card = Create("Frame", {
        Parent           = parent,
        Size             = UDim2.new(0, cardW, 0, cardH),
        Position         = startPos,
        AnchorPoint      = anchor,
        BackgroundColor3 = COLORS.Surface,
        BorderSizePixel  = 0,
    })
    AddCorner(card, 8)
    AddStroke(card, COLORS.Border, 1)

    local iconFrame = Create("Frame", {
        Parent           = card,
        Size             = UDim2.new(0, 36, 0, 36),
        Position         = UDim2.new(0, 12, 0.5, 0),
        AnchorPoint      = Vector2.new(0, 0.5),
        BackgroundColor3 = COLORS.SurfaceAlt,
        BorderSizePixel  = 0,
    })
    AddCorner(iconFrame, 18)

    local iconLabel = Create("TextLabel", {
        Parent           = iconFrame,
        Size             = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        Text             = icon or "!",
        TextColor3       = COLORS.TextPrimary,
        Font             = FONTS.Bold,
        TextSize         = 16,
        TextXAlignment   = Enum.TextXAlignment.Center,
        TextYAlignment   = Enum.TextYAlignment.Center,
    })

    local titleLabel = Create("TextLabel", {
        Parent           = card,
        Size             = UDim2.new(1, -72, 0, 20),
        Position         = UDim2.new(0, 58, 0, 14),
        BackgroundTransparency = 1,
        Text             = title,
        TextColor3       = COLORS.TextPrimary,
        Font             = FONTS.Bold,
        TextSize         = 13,
        TextXAlignment   = Enum.TextXAlignment.Left,
    })

    local msgLabel = Create("TextLabel", {
        Parent           = card,
        Size             = UDim2.new(1, -72, 0, 32),
        Position         = UDim2.new(0, 58, 0, 34),
        BackgroundTransparency = 1,
        Text             = message,
        TextColor3       = COLORS.TextSecondary,
        Font             = FONTS.Regular,
        TextSize          = 11,
        TextXAlignment   = Enum.TextXAlignment.Left,
        TextWrapped      = true,
    })

    local closeBtn = Create("TextButton", {
        Parent           = card,
        Size             = UDim2.new(0, 20, 0, 20),
        Position         = UDim2.new(1, -8, 0, 8),
        AnchorPoint      = Vector2.new(1, 0),
        BackgroundTransparency = 1,
        Text             = "x",
        TextColor3       = COLORS.TextMuted,
        Font             = FONTS.Bold,
        TextSize         = 12,
    })

    card.BackgroundTransparency = 1
    Tween(card, EASING.Smooth, { Position = endPos, BackgroundTransparency = 0 })

    local function dismiss()
        Tween(card, EASING.Smooth, { Position = startPos, BackgroundTransparency = 1 })
        task.delay(0.3, function()
            card:Destroy()
        end)
    end

    closeBtn.MouseButton1Click:Connect(dismiss)

    task.delay(duration, function()
        if card and card.Parent then
            dismiss()
        end
    end)
end

function RiftUI.new(config)
    config = config or {}
    local self = setmetatable({}, RiftUI)

    self.Title    = config.title    or "Rift UI Library"
    self.Subtitle = config.subtitle or "Clean modern Roblox UI"
    self.IconText = config.text     or "Rift"
    self.Size     = config.size     or Vector2.new(720, 480)
    self.AccentColor = config.color or COLORS.White
    self.Tabs     = {}
    self.ActiveTab = nil
    self._connections = {}

    self:_BuildScreenGui()
    self:_BuildMainWindow()
    self:_BuildDraggableIcon()
    self:_BuildProfileSection()
    self:_BuildConfirmModal()
    self:_SetupDrag()
    self:_StartPingUpdate()

    return self
end

function RiftUI:_BuildDraggableIcon()
    self.IconFrame = Create("Frame", {
        Parent           = self.ScreenGui,
        Size             = UDim2.new(0, 56, 0, 56),
        Position         = UDim2.new(0.05, 0, 0.85, 0),
        BackgroundColor3 = COLORS.Surface,
        BorderSizePixel  = 0,
        Visible          = false,
    })
    AddCorner(self.IconFrame, 28)
    AddStroke(self.IconFrame, COLORS.Border, 1)

    Create("TextLabel", {
        Parent           = self.IconFrame,
        Size             = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        Text             = self.IconText,
        TextColor3       = COLORS.TextPrimary,
        Font             = FONTS.Bold,
        TextSize         = 12,
        TextXAlignment   = Enum.TextXAlignment.Center,
        TextYAlignment   = Enum.TextYAlignment.Center,
    })

    local iconBtn = Create("TextButton", {
        Parent           = self.IconFrame,
        Size             = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        Text             = "",
    })

    self.IconFrame.MouseEnter:Connect(function()
        Tween(self.IconFrame, EASING.Fast, {
            BackgroundColor3 = COLORS.Hover,
            Size             = UDim2.new(0, 60, 0, 60),
        })
    end)
    self.IconFrame.MouseLeave:Connect(function()
        Tween(self.IconFrame, EASING.Fast, {
            BackgroundColor3 = COLORS.Surface,
            Size             = UDim2.new(0, 56, 0, 56),
        })
    end)

    iconBtn.MouseButton1Click:Connect(function()
        self:_ShowWindow()
    end)

    local dragging, dragStart, startPos = false, nil, nil

    iconBtn.MouseButton1Down:Connect(function()
        dragging  = true
        dragStart = UserInputService:GetMouseLocation()
        startPos  = self.IconFrame.Position
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)
    RunService.Heartbeat:Connect(function()
        if dragging then
            local delta = UserInputService:GetMouseLocation() - dragStart
            self.IconFrame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end

    self.MainFrame.Visible = true

    self.MainFrame.Size = UDim2.new(0, self.Size.X * 0.96, 0, self.Size.Y * 0.96)
    self.MainFrame.BackgroundTransparency = 1

        BackgroundTransparency = 0,
        Size = UDim2.new(0, self.Size.X, 0, self.Size.Y),
    })
end

function RiftUI:_SetupDrag()
    local dragging, dragStart, startPos = false, nil, nil

    self.Header.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging  = true
            dragStart = UserInputService:GetMouseLocation()
            startPos  = self.MainFrame.Position
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)
    RunService.Heartbeat:Connect(function()
        if dragging then
            local delta = UserInputService:GetMouseLocation() - dragStart
            local np = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
            self.MainFrame.Position = np
            self.ProfileFrame.Position = UDim2.new(
                np.X.Scale, np.X.Offset,
                np.Y.Scale, np.Y.Offset + self.Size.Y + 10
            )
        end
    end)
end

function RiftUI:_StartPingUpdate()
    task.spawn(function()
        while self.ScreenGui and self.ScreenGui.Parent do
            local ping = LocalPlayer:GetNetworkPing and math.round(LocalPlayer:GetNetworkPing() * 1000) or 0
            local color = COLORS.Green
            if ping > 100 then color = Color3.fromRGB(230, 180, 60) end
            if ping > 200 then color = COLORS.Red end
            if self.PingLabel and self.PingLabel.Parent then
                self.PingLabel.Text = "Ping  " .. ping .. " ms"
                self.PingLabel.TextColor3 = color
            end
            task.wait(2)
        end
    end)
end

function RiftUI:_SelectTab(tab)
    if self.ActiveTab then
        Tween(self.ActiveTab.Label, EASING.Smooth, { TextColor3 = COLORS.TextMuted })
        Tween(self.ActiveTab.ContentFrame, EASING.Fast, { })
        self.ActiveTab.ContentFrame.Visible = false
    end

    self.ActiveTab = tab
    tab.ContentFrame.Visible = true

    Tween(tab.Label, EASING.Smooth, { TextColor3 = COLORS.White })

    local targetY = (tab.Index - 1) * 34 + 6
    Tween(self.TabIndicator, EASING.Spring, {
        Position = UDim2.new(0, 0, 0, targetY),
        Size     = UDim2.new(0, 3, 0, 30),
    })
end

    function section:AddSlider(options)
        options = options or {}
        local name     = options.name     or "Slider"
        local min      = options.min      or 0
        local max      = options.max      or 100
        local default  = options.default  or min
        local suffix   = options.suffix   or ""
        local callback = options.callback or function() end

        local value = math.clamp(default, min, max)
        local row = AddRow(54)
        AddPadding(row, 0, 14, 0, 14)

        local nameLabel = Create("TextLabel", {
            Parent           = row,
            Size             = UDim2.new(0.6, 0, 0, 20),
            Position         = UDim2.new(0, 0, 0, 8),
            BackgroundTransparency = 1,
            Text             = name,
            TextColor3       = COLORS.TextPrimary,
            Font             = FONTS.Medium,
            TextSize         = 12,
            TextXAlignment   = Enum.TextXAlignment.Left,
        })

        local valLabel = Create("TextLabel", {
            Parent           = row,
            Size             = UDim2.new(0.4, 0, 0, 20),
            Position         = UDim2.new(0.6, 0, 0, 8),
            BackgroundTransparency = 1,
            Text             = tostring(value) .. suffix,
            TextColor3       = COLORS.TextSecondary,
            Font             = FONTS.Medium,
            TextSize          = 11,
            TextXAlignment   = Enum.TextXAlignment.Right,
        })

        local track = Create("Frame", {
            Parent           = row,
            Size             = UDim2.new(1, 0, 0, 4),
            Position         = UDim2.new(0, 0, 1, -16),
            BackgroundColor3 = COLORS.Border,
            BorderSizePixel  = 0,
        })
        AddCorner(track, 2)

        local pct = (value - min) / (max - min)
        local fill = Create("Frame", {
            Parent           = track,
            Size             = UDim2.new(pct, 0, 1, 0),
            BackgroundColor3 = COLORS.White,
            BorderSizePixel  = 0,
        })
        AddCorner(fill, 2)

        local handle = Create("Frame", {
            Parent           = track,
            Size             = UDim2.new(0, 12, 0, 12),
            Position         = UDim2.new(pct, -6, 0.5, 0),
            AnchorPoint      = Vector2.new(0, 0.5),
            BackgroundColor3 = COLORS.White,
            BorderSizePixel  = 0,
        })
        AddCorner(handle, 6)

        local sliding = false
        local trackBtn = Create("TextButton", {
            Parent              = track,
            Size                = UDim2.new(1, 0, 0, 20),
            Position            = UDim2.new(0, 0, 0.5, 0),
            AnchorPoint         = Vector2.new(0, 0.5),
            BackgroundTransparency = 1,
            Text                = "",
        })

        local function update(mx)
            local absPos  = track.AbsolutePosition.X
            local absSize = track.AbsoluteSize.X
            local ratio   = math.clamp((mx - absPos) / absSize, 0, 1)
            value = math.round(min + ratio * (max - min))
            valLabel.Text = tostring(value) .. suffix
            Tween(fill, EASING.Linear, { Size = UDim2.new(ratio, 0, 1, 0) })
            Tween(handle, EASING.Linear, { Position = UDim2.new(ratio, -6, 0.5, 0) })
            callback(value)
        end

        trackBtn.MouseButton1Down:Connect(function()
            sliding = true
            Tween(handle, EASING.Fast, { Size = UDim2.new(0, 14, 0, 14) })
        end)
        UserInputService.InputEnded:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.MouseButton1 then
                sliding = false
                Tween(handle, EASING.Fast, { Size = UDim2.new(0, 12, 0, 12) })
            end
        end)
        RunService.Heartbeat:Connect(function()
            if sliding then
                update(UserInputService:GetMouseLocation().X)
            end
        end)

        return {
            SetValue = function(v)
                value = math.clamp(v, min, max)
                local r = (value - min) / (max - min)
                valLabel.Text = tostring(value) .. suffix
                Tween(fill, EASING.Smooth, { Size = UDim2.new(r, 0, 1, 0) })
                Tween(handle, EASING.Smooth, { Position = UDim2.new(r, -6, 0.5, 0) })
            end,
            GetValue = function() return value end,
        }
    end

            Tween(stroke, EASING.Fast, { Color = COLORS.White })
        end)
        box.FocusLost:Connect(function(enter)
            Tween(stroke, EASING.Fast, { Color = COLORS.Border })
            if enter then callback(box.Text) end
        end)

        return {
            SetValue = function(v) box.Text = v end,
            GetValue = function() return box.Text end,
        }
    end

    function section:AddKeybind(options)
        options = options or {}
        local name     = options.name     or "Keybind"
        local default  = options.default  or Enum.KeyCode.Unknown
        local callback = options.callback or function() end

        local key      = default
        local listening = false
        local row = AddRow(44)
        AddPadding(row, 0, 14, 0, 14)

        Create("TextLabel", {
            Parent           = row,
            Size             = UDim2.new(0.6, 0, 1, 0),
            BackgroundTransparency = 1,
            Text             = name,
            TextColor3       = COLORS.TextPrimary,
            Font             = FONTS.Medium,
            TextSize         = 12,
            TextXAlignment   = Enum.TextXAlignment.Left,
        })

        local keyBox = Create("TextButton", {
            Parent           = row,
            Size             = UDim2.new(0, 80, 0, 26),
            Position         = UDim2.new(1, 0, 0.5, 0),
            AnchorPoint      = Vector2.new(1, 0.5),
            BackgroundColor3 = COLORS.Surface,
            BorderSizePixel  = 0,
            Text             = key.Name,
            TextColor3       = COLORS.TextPrimary,
            Font             = FONTS.Mono,
            TextSize         = 11,
        })
        AddCorner(keyBox, 4)
        AddStroke(keyBox, COLORS.Border, 1)

        keyBox.MouseButton1Click:Connect(function()
            listening = true
            keyBox.Text = "..."
            Tween(keyBox, EASING.Fast, {
                BackgroundColor3 = COLORS.SurfaceAlt,
                TextColor3       = COLORS.TextSecondary,
            })
        end)

        UserInputService.InputBegan:Connect(function(input, gpe)
            if listening and input.UserInputType == Enum.UserInputType.Keyboard then
                listening = false
                key = input.KeyCode
                keyBox.Text = key.Name
                Tween(keyBox, EASING.Fast, {
                    BackgroundColor3 = COLORS.Surface,
                    TextColor3       = COLORS.TextPrimary,
                })
                callback(key)
            end
        end)

        return {
            GetValue = function() return key end,
        }
    end

    function section:AddColorPicker(options)
        options = options or {}
        local name     = options.name     or "Color Picker"
        local default  = options.default  or Color3.fromRGB(255, 100, 100)
        local callback = options.callback or function() end

        local color = default
        local open  = false
        local row   = AddRow(44)
        AddPadding(row, 0, 14, 0, 14)

        Create("TextLabel", {
            Parent           = row,
            Size             = UDim2.new(0.7, 0, 1, 0),
            BackgroundTransparency = 1,
            Text             = name,
            TextColor3       = COLORS.TextPrimary,
            Font             = FONTS.Medium,
            TextSize         = 12,
            TextXAlignment   = Enum.TextXAlignment.Left,
        })

        local preview = Create("Frame", {
            Parent           = row,
            Size             = UDim2.new(0, 28, 0, 28),
            Position         = UDim2.new(1, 0, 0.5, 0),
            AnchorPoint      = Vector2.new(1, 0.5),
            BackgroundColor3 = color,
            BorderSizePixel  = 0,
        })
        AddCorner(preview, 6)
        AddStroke(preview, COLORS.Border, 1)

        local pickerFrame = Create("Frame", {
            Parent           = row,
            Size             = UDim2.new(0, 200, 0, 0),
            Position         = UDim2.new(1, 0, 1, 6),
            AnchorPoint      = Vector2.new(1, 0),
            BackgroundColor3 = COLORS.Surface,
            BorderSizePixel  = 0,
            ClipsDescendants = true,
            Visible          = false,
            ZIndex           = 30,
        })
        AddCorner(pickerFrame, 8)
        AddStroke(pickerFrame, COLORS.Border, 1)

        local hue, sat, val = Color3.toHSV(color)

        local satValGrad = Create("ImageLabel", {
            Parent           = pickerFrame,
            Size             = UDim2.new(1, -12, 0, 120),
            Position         = UDim2.new(0, 6, 0, 8),
            BackgroundColor3 = Color3.fromHSV(hue, 1, 1),
            BorderSizePixel  = 0,
            Image            = "rbxassetid://4155801252",
            ZIndex           = 31,
        })
        AddCorner(satValGrad, 4)

        local selector = Create("Frame", {
            Parent           = satValGrad,
            Size             = UDim2.new(0, 10, 0, 10),
            Position         = UDim2.new(sat, -5, 1 - val, -5),
            BackgroundColor3 = COLORS.White,
            BorderSizePixel  = 0,
            ZIndex           = 32,
        })
        AddCorner(selector, 5)

        local hueBar = Create("ImageLabel", {
            Parent           = pickerFrame,
            Size             = UDim2.new(1, -12, 0, 14),
            Position         = UDim2.new(0, 6, 0, 134),
            BackgroundColor3 = COLORS.White,
            Image            = "rbxassetid://4155857328",
            BorderSizePixel  = 0,
            ZIndex           = 31,
        })
        AddCorner(hueBar, 4)

        local hueHandle = Create("Frame", {
            Parent           = hueBar,
            Size             = UDim2.new(0, 4, 1, 4),
            Position         = UDim2.new(hue, -2, 0, -2),
            BackgroundColor3 = COLORS.White,
            BorderSizePixel  = 0,
            ZIndex           = 32,
        })
        AddCorner(hueHandle, 2)
        AddStroke(hueHandle, COLORS.Border, 1)

        local function updateColor()
            color = Color3.fromHSV(hue, sat, val)
            preview.BackgroundColor3 = color
            satValGrad.BackgroundColor3 = Color3.fromHSV(hue, 1, 1)
            callback(color)
        end

        local draggingSV, draggingHue = false, false

        local svBtn = Create("TextButton", {
            Parent              = satValGrad,
            Size                = UDim2.new(1, 0, 1, 0),
            BackgroundTransparency = 1,
            Text                = "",
            ZIndex              = 33,
        })
        svBtn.MouseButton1Down:Connect(function() draggingSV = true end)
        UserInputService.InputEnded:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.MouseButton1 then
                draggingSV = false
                draggingHue = false
            end
        end)
        RunService.Heartbeat:Connect(function()
            if draggingSV then
                local mp   = UserInputService:GetMouseLocation()
                local abs  = satValGrad.AbsolutePosition
                local size = satValGrad.AbsoluteSize
                sat = math.clamp((mp.X - abs.X) / size.X, 0, 1)
                val = 1 - math.clamp((mp.Y - abs.Y) / size.Y, 0, 1)
                selector.Position = UDim2.new(sat, -5, 1 - val, -5)
                updateColor()
            end
            if draggingHue then
                local mp   = UserInputService:GetMouseLocation()
                local abs  = hueBar.AbsolutePosition
                local size = hueBar.AbsoluteSize
                hue = math.clamp((mp.X - abs.X) / size.X, 0, 1)
                hueHandle.Position = UDim2.new(hue, -2, 0, -2)
                updateColor()
            end
        end)

        local hueBtn = Create("TextButton", {
            Parent              = hueBar,
            Size                = UDim2.new(1, 0, 1, 0),
            BackgroundTransparency = 1,
            Text                = "",
            ZIndex              = 33,
        })
        hueBtn.MouseButton1Down:Connect(function() draggingHue = true end)

        local previewBtn = Create("TextButton", {
            Parent              = preview,
            Size                = UDim2.new(1, 0, 1, 0),
            BackgroundTransparency = 1,
            Text                = "",
            ZIndex              = 10,
        })
        previewBtn.MouseButton1Click:Connect(function()
            open = not open
            if open then
                pickerFrame.Visible = true
                Tween(pickerFrame, EASING.Spring, { Size = UDim2.new(0, 200, 0, 162) })
            else
                Tween(pickerFrame, EASING.Smooth, { Size = UDim2.new(0, 200, 0, 0) })
                task.delay(0.25, function() pickerFrame.Visible = false end)
            end
        end)

        return {
            GetValue = function() return color end,
            SetValue = function(c)
                color = c
                hue, sat, val = Color3.toHSV(c)
                preview.BackgroundColor3 = c
                satValGrad.BackgroundColor3 = Color3.fromHSV(hue, 1, 1)
                selector.Position = UDim2.new(sat, -5, 1 - val, -5)
                hueHandle.Position = UDim2.new(hue, -2, 0, -2)
            end,
        }
    end

    function section:AddImage(options)
        options = options or {}
        local name  = options.name  or "Image"
        local image = options.image or ""
        local onClick = options.onClick or nil

        local row = AddRow(80)
        AddPadding(row, 8, 14, 8, 14)

        Create("TextLabel", {
            Parent           = row,
            Size             = UDim2.new(1, 0, 0, 16),
            BackgroundTransparency = 1,
            Text             = name,
            TextColor3       = COLORS.TextPrimary,
            Font             = FONTS.Medium,
            TextSize         = 12,
            TextXAlignment   = Enum.TextXAlignment.Left,
        })

        local imgFrame = Create("Frame", {
            Parent           = row,
            Size             = UDim2.new(1, 0, 1, -20),
            Position         = UDim2.new(0, 0, 0, 20),
            BackgroundColor3 = COLORS.Background,
            BorderSizePixel  = 0,
        })
        AddCorner(imgFrame, 6)
        AddStroke(imgFrame, COLORS.Border, 1)

        local imgLabel = Create("ImageLabel", {
            Parent           = imgFrame,
            Size             = UDim2.new(1, -4, 1, -4),
            Position         = UDim2.new(0, 2, 0, 2),
            BackgroundTransparency = 1,
            Image            = image,
            ScaleType        = Enum.ScaleType.Fit,
        })
        AddCorner(imgLabel, 4)

        if onClick then
            local btn = Create("TextButton", {
                Parent              = imgFrame,
                Size                = UDim2.new(1, 0, 1, 0),
                BackgroundTransparency = 1,
                Text                = "",
            })
            btn.MouseButton1Down:Connect(function()
                Tween(imgFrame, EASING.Fast, { Size = UDim2.new(0.98, 0, 0.96, -20) })
            end)
            btn.MouseButton1Up:Connect(function()
                Tween(imgFrame, EASING.Spring, { Size = UDim2.new(1, 0, 1, -20) })
            end)
            btn.MouseButton1Click:Connect(onClick)
        end
    end

    return section
end

    RiftUI.Notify(self, options)
end
