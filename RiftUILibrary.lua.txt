local function Create(instanceType, properties)
    local instance = Instance.new(instanceType)
    for prop, value in pairs(properties or {}) do
        instance[prop] = value
    end
    return instance
end

local function Tween(instance, properties, duration, easingStyle, easingDirection)
    duration = duration or Config.AnimationDuration
    easingStyle = easingStyle or Config.EasingStyle
    easingDirection = easingDirection or Config.EasingDirection
    local tweenInfo = TweenInfo.new(duration, easingStyle, easingDirection)
    local tween = TweenService:Create(instance, tweenInfo, properties)
    tween:Play()
    return tween
end

local function CreateStroke(parent, color, thickness)
    return Create("UIStroke", {
        Parent = parent,
        Color = color or Config.Colors.Stroke,
        Thickness = thickness or 1,
        Transparency = 0.8
    })
end

local function CreateCorner(parent, radius)
    return Create("UICorner", {
        Parent = parent,
        CornerRadius = UDim.new(0, radius or 6)
    })
end

local function CreateShadow(parent, size)
    return Create("ImageLabel", {
        Parent = parent,
        BackgroundTransparency = 1,
        Image = "rbxassetid://5554236805",
        ImageColor3 = Color3.fromRGB(0, 0, 0),
        ImageTransparency = 0.6,
        Position = UDim2.new(0, -size/2, 0, -size/2),
        Size = UDim2.new(1, size, 1, size),
        ZIndex = parent.ZIndex - 1
    })
end

function Rift:Notify(options)

        Parent = self.NotificationGui,

        Parent = notifFrame,
        BackgroundTransparency = 1,

        ImageTransparency = 1,
        ZIndex = 101
    })
    
    local textLabel = Create("TextLabel", {
        Parent = notifFrame,
        BackgroundTransparency = 1,
        Position = UDim2.new(0, icon ~= "" and 65 or 15, 0, 0),
        Size = UDim2.new(1, icon ~= "" and -80 or -30, 1, 0),
        Font = Enum.Font.GothamMedium,
        Text = text,
        TextColor3 = Config.Colors.Text,
        TextSize = 14,
        TextWrapped = true,
        TextTransparency = 1,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 101
    })
    
    local closeBtn = Create("TextButton", {
        Parent = notifFrame,
        BackgroundTransparency = 1,
        Position = UDim2.new(1, -30, 0, 5),
        Size = UDim2.new(0, 25, 0, 25),
        Font = Enum.Font.GothamBold,
        Text = "×",
        TextColor3 = Config.Colors.SubText,
        TextSize = 20,
        ZIndex = 102
    })
    
    local progressBar = Create("Frame", {
        Parent = notifFrame,
        BackgroundColor3 = Config.Colors.Accent,
        BorderSizePixel = 0,
        Position = UDim2.new(0, 0, 1, -2),
        Size = UDim2.new(0, 0, 0, 2),
        ZIndex = 101
    })
    
    Tween(notifFrame, {BackgroundTransparency = 0, Position = positions[position]}, 0.4, Config.SpringEasing)
    Tween(textLabel, {TextTransparency = 0}, 0.3)
    Tween(iconLabel, {ImageTransparency = 0}, 0.3)
    Tween(progressBar, {Size = UDim2.new(1, 0, 0, 2)}, duration, Enum.EasingStyle.Linear)
    
    local function close()
        Tween(notifFrame, {BackgroundTransparency = 1, Position = notifFrame.Position + UDim2.new(0, 50, 0, 0)}, 0.3)
        Tween(textLabel, {TextTransparency = 1}, 0.2)
        Tween(iconLabel, {ImageTransparency = 1}, 0.2)
        task.wait(0.3)
        notifFrame:Destroy()
    end
    
    closeBtn.MouseButton1Click:Connect(close)
    task.delay(duration, close)
end

    function Window:CreateTab(tabName)
        local Tab = {}

            Parent = TabList,
            BackgroundColor3 = Config.Colors.DarkBackground,
            BorderSizePixel = 0,
            Size = UDim2.new(1, -10, 0, 32),
            Position = UDim2.new(0, 5, 0, 0),
            Font = Enum.Font.GothamMedium,
            Text = tabName,

        })
        
        local TabContent = Create("ScrollingFrame", {
            Parent = ContentContainer,
            BackgroundTransparency = 1,

            ScrollBarThickness = 2,

            Visible = false,
            CanvasSize = UDim2.new(0, 0, 0, 0)
        })
        
        local TabContentLayout = Create("UIListLayout", {
            Parent = TabContent,
            Padding = UDim.new(0, 10),
            SortOrder = Enum.SortOrder.LayoutOrder
        })
        
        TabContentLayout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
            TabContent.CanvasSize = UDim2.new(0, 0, 0, TabContentLayout.AbsoluteContentSize.Y + 20)
        end)
        
        Tab.TabButton = TabButton
        Tab.TabContent = TabContent
        
        table.insert(self.Tabs, Tab)
        
        TabButton.MouseButton1Click:Connect(function()
            if Window.ActiveTab == Tab then return end
            
            local btnPos = TabButton.AbsolutePosition.Y - TabList.AbsolutePosition.Y + TabList.CanvasPosition.Y
            Tween(ActiveIndicator, {Position = UDim2.new(0, 0, 0, 10 + btnPos)}, 0.3, Config.SpringEasing)
            
            if Window.ActiveTab then
                Tween(Window.ActiveTab.TabButton, {TextColor3 = Config.Colors.SubText}, Config.FastDuration)
                Window.ActiveTab.TabContent.Visible = false
            end
            
            Window.ActiveTab = Tab
            TabContent.Visible = true
            Tween(TabButton, {TextColor3 = Config.Colors.Text}, Config.FastDuration)
        end)
        
        TabButton.MouseEnter:Connect(function()
            if Window.ActiveTab ~= Tab then
                Tween(TabButton, {TextColor3 = Config.Colors.Text}, Config.FastDuration)
            end
        end)
        
        TabButton.MouseLeave:Connect(function()
            if Window.ActiveTab ~= Tab then
                Tween(TabButton, {TextColor3 = Config.Colors.SubText}, Config.FastDuration)
            end

        function Tab:CreateSlider(options)
            options = options or {}
            local text = options.text or "Slider"
            local min = options.min or 0
            local max = options.max or 100
            local default = options.default or min
            local callback = options.callback or function() end
            
            local SliderFrame = Create("Frame", {
                Parent = TabContent,
                BackgroundColor3 = Config.Colors.LightBackground,
                BorderSizePixel = 0,
                Size = UDim2.new(1, -10, 0, 55)
            })
            CreateCorner(SliderFrame, 8)
            
            local Label = Create("TextLabel", {
                Parent = SliderFrame,
                BackgroundTransparency = 1,
                Position = UDim2.new(0, 15, 0, 8),
                Size = UDim2.new(0.5, 0, 0, 20),
                Font = Enum.Font.GothamMedium,
                Text = text,
                TextColor3 = Config.Colors.Text,
                TextSize = 14,
                TextXAlignment = Enum.TextXAlignment.Left
            })
            
            local ValueLabel = Create("TextLabel", {
                Parent = SliderFrame,
                BackgroundTransparency = 1,
                Position = UDim2.new(1, -60, 0, 8),
                Size = UDim2.new(0, 45, 0, 20),
                Font = Enum.Font.GothamBold,
                Text = tostring(default),
                TextColor3 = accentColor,
                TextSize = 14,
                TextXAlignment = Enum.TextXAlignment.Right
            })
            
            local Track = Create("Frame", {
                Parent = SliderFrame,
                BackgroundColor3 = Config.Colors.DarkBackground,
                BorderSizePixel = 0,
                Position = UDim2.new(0, 15, 0, 38),
                Size = UDim2.new(1, -30, 0, 4)
            })
            CreateCorner(Track, 2)
            
            local Fill = Create("Frame", {
                Parent = Track,
                BackgroundColor3 = accentColor,
                BorderSizePixel = 0,
                Size = UDim2.new((default - min) / (max - min), 0, 1, 0)
            })
            CreateCorner(Fill, 2)
            
            local Handle = Create("Frame", {
                Parent = Track,
                BackgroundColor3 = Config.Colors.Text,
                BorderSizePixel = 0,
                Position = UDim2.new((default - min) / (max - min), -8, 0.5, -8),
                Size = UDim2.new(0, 16, 0, 16),
                ZIndex = 2
            })
            CreateCorner(Handle, 8)
            
            local ClickArea = Create("TextButton", {
                Parent = SliderFrame,
                BackgroundTransparency = 1,
                Size = UDim2.new(1, 0, 1, 0),
                Text = ""
            })
            
            local dragging = false
            local value = default
            
            local function updateSlider(input)
                local pos = math.clamp((input.Position.X - Track.AbsolutePosition.X) / Track.AbsoluteSize.X, 0, 1)
                value = math.floor(min + (max - min) * pos)
                ValueLabel.Text = tostring(value)
                Tween(Fill, {Size = UDim2.new(pos, 0, 1, 0)}, 0.05)
                Tween(Handle, {Position = UDim2.new(pos, -8, 0.5, -8)}, 0.05)
                callback(value)
            end
            
            ClickArea.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 then
                    dragging = true
                    updateSlider(input)
                end
            end)
            
            UserInputService.InputChanged:Connect(function(input)
                if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
                    updateSlider(input)
                end
            end)
            
            UserInputService.InputEnded:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 then
                    dragging = false
                end
            end)
            
            Handle.MouseEnter:Connect(function()
                Tween(Handle, {Size = UDim2.new(0, 20, 0, 20), Position = Handle.Position - UDim2.new(0, 2, 0, 2)}, Config.FastDuration)
            end)
            
            Handle.MouseLeave:Connect(function()
                if not dragging then
                    Tween(Handle, {Size = UDim2.new(0, 16, 0, 16), Position = Handle.Position + UDim2.new(0, 2, 0, 2)}, Config.FastDuration)
                end
            end)
            
            return {
                Set = function(val)
                    value = math.clamp(val, min, max)
                    local pos = (value - min) / (max - min)
                    ValueLabel.Text = tostring(value)
                    Tween(Fill, {Size = UDim2.new(pos, 0, 1, 0)}, 0.1)
                    Tween(Handle, {Position = UDim2.new(pos, -8, 0.5, -8)}, 0.1)
                    callback(value)
                end,
                Get = function() return value end
            }
        end

            local OptionsLayout = Create("UIListLayout", {
                Parent = OptionsList,
                Padding = UDim.new(0, 2)
            })

            local function updateOptions(filter)
                for _, child in pairs(OptionsList:GetChildren()) do
                    if child:IsA("TextButton") then child:Destroy() end
                end
                
                for _, value in pairs(values) do
                    if not filter or string.find(string.lower(value), string.lower(filter)) then
                        local OptionBtn = Create("TextButton", {
                            Parent = OptionsList,
                            BackgroundColor3 = Config.Colors.DarkBackground,
                            BorderSizePixel = 0,
                            Size = UDim2.new(1, -10, 0, 30),

                            AutoButtonColor = false

                            Tween(OptionBtn, {BackgroundColor3 = Config.Colors.LightBackground}, Config.FastDuration)
                        end)
                        
                        OptionBtn.MouseLeave:Connect(function()
                            Tween(OptionBtn, {BackgroundColor3 = Config.Colors.DarkBackground}, Config.FastDuration)
                        end)
                        
                        OptionBtn.MouseButton1Click:Connect(function()
                            selected = value
                            Label.Text = text .. ": " .. selected
                            callback(selected)
                            isOpen = false
                            Tween(Arrow, {Rotation = 0}, Config.FastDuration)
                            Tween(ListFrame, {Size = UDim2.new(1, -10, 0, 0)}, 0.2)
                            task.wait(0.2)
                            ListFrame.Visible = false
                            Tween(DropdownFrame, {Size = UDim2.new(1, -10, 0, 40)}, 0.2)
                        end)
                    end
                end
                
                OptionsList.CanvasSize = UDim2.new(0, 0, 0, OptionsLayout.AbsoluteContentSize.Y + 10)
            end
            
            updateOptions()
            
            local ClickArea = Create("TextButton", {
                Parent = DropdownFrame,
                BackgroundTransparency = 1,
                Size = UDim2.new(1, 0, 0, 40),
                Text = ""
            })
            
            ClickArea.MouseButton1Click:Connect(function()
                isOpen = not isOpen
                if isOpen then
                    ListFrame.Visible = true
                    Tween(DropdownFrame, {Size = UDim2.new(1, -10, 0, 200)}, 0.3)
                    Tween(ListFrame, {Size = UDim2.new(1, -10, 0, 150)}, 0.3)
                    Tween(Arrow, {Rotation = 180}, Config.FastDuration)
                else
                    Tween(Arrow, {Rotation = 0}, Config.FastDuration)
                    Tween(ListFrame, {Size = UDim2.new(1, -10, 0, 0)}, 0.2)
                    task.wait(0.2)
                    ListFrame.Visible = false
                    Tween(DropdownFrame, {Size = UDim2.new(1, -10, 0, 40)}, 0.2)
                end
            end)
            
            SearchBox:GetPropertyChangedSignal("Text"):Connect(function()
                updateOptions(SearchBox.Text)
            end)
            
            return {
                Set = function(val)
                    selected = val
                    Label.Text = text .. ": " .. selected
                    callback(selected)
                end,
                Get = function() return selected end,
                Refresh = function(newValues)
                    values = newValues
                    updateOptions(SearchBox.Text)
                end
            }
        end

                BackgroundColor3 = Color3.fromRGB(255, 255, 255),
                BorderSizePixel = 0,
                Size = UDim2.new(1, 0, 1, 0),
                ZIndex = 52
            })
            
            local WhiteGradient = Create("UIGradient", {
                Parent = WhiteOverlay,
                Color = ColorSequence.new({
                    ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 255, 255)),
                    ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 255, 255))
                }),
                Transparency = NumberSequence.new({
                    NumberSequenceKeypoint.new(0, 0),
                    NumberSequenceKeypoint.new(1, 1)
                })
            })
            
            local BlackOverlay = Create("Frame", {
                Parent = SaturationFrame,
                BackgroundColor3 = Color3.fromRGB(0, 0, 0),
                BorderSizePixel = 0,
                Size = UDim2.new(1, 0, 1, 0),
                ZIndex = 53
            })
            
            local BlackGradient = Create("UIGradient", {
                Parent = BlackOverlay,
                Rotation = 90,
                Color = ColorSequence.new({
                    ColorSequenceKeypoint.new(0, Color3.fromRGB(0, 0, 0)),
                    ColorSequenceKeypoint.new(1, Color3.fromRGB(0, 0, 0))
                }),
                Transparency = NumberSequence.new({
                    NumberSequenceKeypoint.new(0, 1),
                    NumberSequenceKeypoint.new(1, 0)
                })
            })
            
            local isOpen = false
            local currentColor = default
            
            ColorPreview.MouseButton1Click:Connect(function()
                isOpen = not isOpen
                if isOpen then
                    PickerFrame.Visible = true
                    Tween(ColorFrame, {Size = UDim2.new(1, -10, 0, 210)}, 0.3)
                    Tween(PickerFrame, {Size = UDim2.new(1, -10, 0, 170)}, 0.3)
                else
                    Tween(PickerFrame, {Size = UDim2.new(1, -10, 0, 0)}, 0.2)
                    task.wait(0.2)
                    PickerFrame.Visible = false
                    Tween(ColorFrame, {Size = UDim2.new(1, -10, 0, 40)}, 0.2)
                end
            end)
            
            return {
                Set = function(color)
                    currentColor = color
                    ColorPreview.BackgroundColor3 = color
                    callback(color)
                end,
                Get = function() return currentColor end
            }
        end
        
        function Tab:CreateImage(options)
            options = options or {}
            local image = options.image or ""
            local imageSize = options.size or UDim2.new(1, -10, 0, 150)
            local callback = options.callback or function() end
            
            local ImageFrame = Create("Frame", {
                Parent = TabContent,
                BackgroundColor3 = Config.Colors.LightBackground,
                BorderSizePixel = 0,
                Size = imageSize
            })
            CreateCorner(ImageFrame, 8)
            
            local ImageLabel = Create("ImageLabel", {
                Parent = ImageFrame,
                BackgroundTransparency = 1,
                Position = UDim2.new(0, 5, 0, 5),
                Size = UDim2.new(1, -10, 1, -10),
                Image = image,
                ScaleType = Enum.ScaleType.Stretch
            })
            CreateCorner(ImageLabel, 6)
            
            local ClickArea = Create("TextButton", {
                Parent = ImageFrame,
                BackgroundTransparency = 1,
                Size = UDim2.new(1, 0, 1, 0),
                Text = ""
            })
            
            ClickArea.MouseButton1Click:Connect(function()
                Tween(ImageFrame, {Size = UDim2.new(imageSize.X.Scale * 0.98, imageSize.X.Offset * 0.98, 0, imageSize.Y.Offset * 0.98)}, 0.1)
                task.wait(0.1)
                Tween(ImageFrame, {Size = imageSize}, 0.1)
                callback()
            end)
            
            return {
                Set = function(newImage)
                    ImageLabel.Image = newImage
                end
            }
        end
        
        return Tab
    end
    
    return Window
