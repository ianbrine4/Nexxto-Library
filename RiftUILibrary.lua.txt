local function Create(instanceType, properties)
	local instance = Instance.new(instanceType)
	for property, value in pairs(properties or {}) do
		instance[property] = value
	end
	return instance
end

local function Tween(instance, properties, duration, easingStyle, easingDirection)
	local tween = TweenService:Create(instance, TweenInfo.new(duration or 0.3, easingStyle or Enum.EasingStyle.Quart, easingDirection or Enum.EasingDirection.Out), properties)
	tween:Play()
	return tween
end

local function Round(number, decimalPlaces)
	local multiplier = 10 ^ (decimalPlaces or 0)
	return math.floor(number * multiplier + 0.5) / multiplier
end

	function Window:AddTab(name)
		local Tab = {}
		local TabButton = Create("TextButton", {
			Name = name,
			Parent = TabList,
			BackgroundColor3 = Theme.BackgroundSecondary,
			Text = name,
			TextColor3 = Theme.TextSecondary,
			TextSize = 14,
			Font = Enum.Font.Gotham,
			Size = UDim2.new(1, 0, 0, 35),
			AutoButtonColor = false
		})
		
		Create("UICorner", {
			CornerRadius = UDim.new(0, 6),
			Parent = TabButton
		})
		
		local Separator = Create("Frame", {
			Name = "Separator",
			Parent = TabButton,
			BackgroundColor3 = Theme.Gray,
			BorderSizePixel = 0,
			Size = UDim2.new(1, -20, 0, 1),
			Position = UDim2.new(0, 10, 1, -1),
			Visible = false
		})
		
		local Page = Create("ScrollingFrame", {
			Name = name .. "Page",
			Parent = PageContainer,
			BackgroundTransparency = 1,
			Size = UDim2.new(1, 0, 1, 0),
			ScrollBarThickness = 2,
			ScrollBarImageColor3 = Theme.Gray,
			CanvasSize = UDim2.new(0, 0, 0, 0),
			AutomaticCanvasSize = Enum.AutomaticSize.Y,
			Visible = false
		})
		
		Create("UIPadding", {
			Parent = Page,
			PaddingLeft = UDim.new(0, 20),
			PaddingRight = UDim.new(0, 20),
			PaddingTop = UDim.new(0, 20),
			PaddingBottom = UDim.new(0, 20)
		})
		
		Create("UIListLayout", {
			Parent = Page,
			Padding = UDim.new(0, 15),
			SortOrder = Enum.SortOrder.LayoutOrder
		})
		
		if not TabIndicator then
			TabIndicator = Create("Frame", {
				Name = "Indicator",
				Parent = TabButton,
				BackgroundColor3 = Theme.Accent,
				BorderSizePixel = 0,
				Size = UDim2.new(0, 3, 0, 20),
				Position = UDim2.new(0, -10, 0.5, -10)
			})
			Create("UICorner", {
				CornerRadius = UDim.new(0, 2),
				Parent = TabIndicator
			})
		end
		
		TabButton.MouseEnter:Connect(function()
			if CurrentTab ~= Tab then
				Tween(TabButton, {BackgroundColor3 = Theme.BackgroundTertiary}, 0.2)
			end
		end)
		
		TabButton.MouseLeave:Connect(function()
			if CurrentTab ~= Tab then
				Tween(TabButton, {BackgroundColor3 = Theme.BackgroundSecondary}, 0.2)
			end
		end)
		
		TabButton.MouseButton1Click:Connect(function()
			if CurrentTab == Tab then return end
			
			if CurrentTab then
				Tween(CurrentTab.Button, {BackgroundColor3 = Theme.BackgroundSecondary, TextColor3 = Theme.TextSecondary}, 0.3)
				Tween(CurrentTab.Page, {Position = UDim2.new(0, 20, 0, 0), Transparency = 1}, 0.3)
				delay(0.3, function()
					CurrentTab.Page.Visible = false
				end)
			end
			
			TabIndicator.Parent = TabButton
			Tween(TabIndicator, {Position = UDim2.new(0, -10, 0.5, -10)}, 0.3)
			Tween(TabButton, {BackgroundColor3 = Theme.BackgroundTertiary, TextColor3 = Theme.Text}, 0.3)
			
			Page.Position = UDim2.new(0, -20, 0, 0)
			Page.Visible = true
			Tween(Page, {Position = UDim2.new(0, 0, 0, 0)}, 0.3)
			
			CurrentTab = Tab
		end)
		
		Tab.Button = TabButton
		Tab.Page = Page
		
		if not CurrentTab then
			TabButton.MouseButton1Click:Fire()
		end

		function Tab:AddButton(text, callback)
			local Container = Create("Frame", {
				Parent = Page,
				BackgroundTransparency = 1,
				Size = UDim2.new(1, 0, 0, 40),
				LayoutOrder = #Page:GetChildren()
			})
			
			local Button = Create("TextButton", {
				Parent = Container,
				BackgroundColor3 = Theme.BackgroundTertiary,
				Text = text,
				TextColor3 = Theme.Text,
				TextSize = 14,
				Font = Enum.Font.Gotham,
				Size = UDim2.new(1, 0, 1, 0),
				AutoButtonColor = false
			})
			
			Create("UICorner", {
				CornerRadius = UDim.new(0, 6),
				Parent = Button
			})
			
			local Glow = Create("Frame", {
				Name = "Glow",
				Parent = Button,
				BackgroundColor3 = config.color or Theme.Accent,
				BackgroundTransparency = 0.9,
				BorderSizePixel = 0,
				Size = UDim2.new(1, 0, 1, 0),
				Visible = false
			})
			
			Create("UICorner", {
				CornerRadius = UDim.new(0, 6),
				Parent = Glow
			})
			
			Button.MouseEnter:Connect(function()
				Glow.Visible = true
				Tween(Glow, {BackgroundTransparency = 0.7}, 0.2)
				Tween(Button, {BackgroundColor3 = Theme.BackgroundTertiary}, 0.2)
			end)
			
			Button.MouseLeave:Connect(function()
				Tween(Glow, {BackgroundTransparency = 1}, 0.2).Completed:Connect(function()
					Glow.Visible = false
				end)
			end)
			
			Button.MouseButton1Down:Connect(function()
				Tween(Button, {Size = UDim2.new(0.98, 0, 0.95, 0)}, 0.1)
			end)
			
			Button.MouseButton1Up:Connect(function()
				Tween(Button, {Size = UDim2.new(1, 0, 1, 0)}, 0.1)
				if callback then
					callback()
				end
			end)
			
			return Button
		end

					Hue = 1 - pos
					HueCursor.Position = UDim2.new(0.5, 0, pos, 0)
					UpdateColor()
				end
			end)
			
			SaturationFrame.InputBegan:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1 then
					DraggingSat = true
					local x = math.clamp((input.Position.X - SaturationFrame.AbsolutePosition.X) / SaturationFrame.AbsoluteSize.X, 0, 1)
					local y = math.clamp((input.Position.Y - SaturationFrame.AbsolutePosition.Y) / SaturationFrame.AbsoluteSize.Y, 0, 1)
					Sat = x
					Val = 1 - y
					PickerCursor.Position = UDim2.new(x, 0, y, 0)
					UpdateColor()
				end
			end)
			
			UserInputService.InputChanged:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseMovement then
					if DraggingHue then
						local pos = math.clamp((input.Position.Y - HueBar.AbsolutePosition.Y) / HueBar.AbsoluteSize.Y, 0, 1)
						Hue = 1 - pos
						HueCursor.Position = UDim2.new(0.5, 0, pos, 0)
						UpdateColor()
					end
					if DraggingSat then
						local x = math.clamp((input.Position.X - SaturationFrame.AbsolutePosition.X) / SaturationFrame.AbsoluteSize.X, 0, 1)
						local y = math.clamp((input.Position.Y - SaturationFrame.AbsolutePosition.Y) / SaturationFrame.AbsoluteSize.Y, 0, 1)
						Sat = x
						Val = 1 - y
						PickerCursor.Position = UDim2.new(x, 0, y, 0)
						UpdateColor()
					end
				end
			end)
			
			UserInputService.InputEnded:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1 then
					DraggingHue = false
					DraggingSat = false
				end
			end)
			
			return {
				Set = function(color)
					CurrentColor = color
					ColorPreview.BackgroundColor3 = color
					local h, s, v = Color3.toHSV(color)
					Hue, Sat, Val = h, s, v
					HueCursor.Position = UDim2.new(0.5, 0, 1 - h, 0)
					PickerCursor.Position = UDim2.new(s, 0, 1 - v, 0)
					if callback then
						callback(color)
					end
				end,
				Get = function()
					return CurrentColor
				end
			}
		end
		
		function Tab:AddImage(imageId, size)
			local Container = Create("Frame", {
				Parent = Page,
				BackgroundTransparency = 1,
				Size = UDim2.new(1, 0, 0, (size and size.Y) or 150),
				LayoutOrder = #Page:GetChildren()
			})
			
			local Image = Create("ImageLabel", {
				Parent = Container,
				BackgroundColor3 = Theme.BackgroundTertiary,
				Image = imageId or "",
				Size = UDim2.new(0, (size and size.X) or 200, 1, 0),
				Position = UDim2.new(0.5, -((size and size.X) or 200)/2, 0, 0),
				ScaleType = Enum.ScaleType.Fit
			})
			
			Create("UICorner", {
				CornerRadius = UDim.new(0, 8),
				Parent = Image
			})
			
			Create("UIStroke", {
				Color = Theme.Stroke,
				Thickness = 1,
				Parent = Image
			})

				Tween(Image, {Size = UDim2.new(0, ((size and size.X) or 200) * 1.02, 1.02, 0)}, 0.2)

				Tween(Image, {Size = UDim2.new(0, (size and size.X) or 200, 1, 0)}, 0.2)
			end)

		end
		
		table.insert(Pages, Tab)
		return Tab
	end
